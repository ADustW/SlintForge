cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SlintForgeCppConsumer LANGUAGES CXX)
    set(SLINT_STYLE fluent CACHE STRING "Slint widget style")
    if(MSVC)
        add_compile_options(/utf-8)
    endif()
    find_package(Slint 1.18.0 EXACT CONFIG REQUIRED)
    find_package(SlintForge 0.1.1 CONFIG REQUIRED)
endif()

add_executable(slintforge_example_cpp main.cpp)
if(MSVC)
    target_compile_options(slintforge_example_cpp PRIVATE /UNDEBUG)
endif()
target_link_libraries(slintforge_example_cpp PRIVATE Slint::Slint SlintForge::Sdk)
slintforge_target_sources(slintforge_example_cpp ../ui/minimal.slint)
