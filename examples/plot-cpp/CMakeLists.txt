cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SlintForgePlotExample LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 20)
if(MSVC)
    add_compile_options(/utf-8)
endif()
set(SLINT_STYLE fluent CACHE STRING "Slint style")
if(NOT TARGET SlintForge::Sdk)
    find_package(SlintForge 0.1.1 CONFIG REQUIRED)
endif()
add_executable(slintforge_plot_example main.cpp)
if(MSVC)
    target_compile_options(slintforge_plot_example PRIVATE /UNDEBUG)
endif()
target_link_libraries(slintforge_plot_example PRIVATE SlintForge::Sdk)
slintforge_target_sources(slintforge_plot_example ../ui/professional-plot.slint)
