cmake_minimum_required(VERSION 3.21)
project(SlintForge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
set(SLINT_STYLE fluent CACHE STRING "Slint widget style")
find_package(Slint 1.18.0 EXACT CONFIG REQUIRED)

include(CTest)
add_subdirectory(sdk/cpp)
add_subdirectory(Gallery/cpp)
add_subdirectory(examples/cpp)
if(BUILD_TESTING)

    target_link_libraries(slintforge_plot_tests PRIVATE SlintForge::Sdk)

    set_tests_properties(slintforge_plot_tests PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")

    target_link_libraries(slintforge_grid_review_tests PRIVATE SlintForge::Sdk)
    add_test(NAME slintforge_grid_review_tests COMMAND slintforge_grid_review_tests)
    set_tests_properties(slintforge_grid_review_tests PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
    add_executable(slintforge_grid_tests tests/cpp/ForgeGridTests.cpp)
    target_link_libraries(slintforge_grid_tests PRIVATE SlintForge::Sdk)
    add_test(NAME slintforge_grid_tests COMMAND slintforge_grid_tests)
    set_tests_properties(slintforge_grid_tests PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}")
    add_executable(slintforge_chart_tests tests/cpp/ForgeChartTests.cpp)
    target_link_libraries(slintforge_chart_tests PRIVATE SlintForge::Sdk)
    add_test(NAME slintforge_chart_tests COMMAND slintforge_chart_tests)

    add_test(NAME slintforge_cpp_tests COMMAND slintforge_cpp_tests)

    target_link_libraries(slintforge_capture_tests PRIVATE SlintForge::Sdk)

    set_tests_properties(slintforge_widgets_consumer slintforge_charts_consumer PROPERTIES WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" TIMEOUT 30)
